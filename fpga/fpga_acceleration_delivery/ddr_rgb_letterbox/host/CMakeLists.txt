cmake_minimum_required(VERSION 3.16)
project(delivery_paired_preprocess LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MIGRATION_ROOT "" CACHE PATH "Existing fpga_v8_migration assets")
set(DELIVERY_PROVIDER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/linux_provider.cpp" CACHE FILEPATH "Real board implementation of delivery::bind_provider")
set(RKNN_RUNTIME "" CACHE FILEPATH "Actual RK3568 librknnrt.so")
if(NOT EXISTS "${MIGRATION_ROOT}/cooperative_paired_bench/rknn_runner.hpp")
  message(FATAL_ERROR "Explicit MIGRATION_ROOT required")
endif()
if(NOT EXISTS "${DELIVERY_PROVIDER_SOURCE}" OR NOT EXISTS "${RKNN_RUNTIME}")
  message(FATAL_ERROR "Real hardware DELIVERY_PROVIDER_SOURCE and RKNN_RUNTIME required; no software fallback")
endif()
find_package(Threads REQUIRED)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
add_executable(paired_preprocess paired_preprocess.cpp ${DELIVERY_PROVIDER_SOURCE})
target_include_directories(paired_preprocess PRIVATE ${MIGRATION_ROOT} ${MIGRATION_ROOT}/current_viewer_native_candidate_02/src ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(paired_preprocess PRIVATE Qt${QT_VERSION_MAJOR}::Core Threads::Threads ${RKNN_RUNTIME})
