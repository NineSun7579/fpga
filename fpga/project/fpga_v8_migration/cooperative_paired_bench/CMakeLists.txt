cmake_minimum_required(VERSION 3.16)
project(cooperative_paired_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(paired_fake fake_test.cpp)
target_link_libraries(paired_fake PRIVATE Threads::Threads)
enable_testing()
add_test(NAME paired_faults COMMAND paired_fake)
option(PAIRED_BUILD_RKNN "Build RK3568 runner; does not execute it" OFF)
set(PAIRED_PLATFORM_BINDING "${CMAKE_CURRENT_SOURCE_DIR}/platform_unbound.cpp" CACHE FILEPATH "Own delegate source implementing bind_platform")
set(RKNN_RUNTIME "" CACHE FILEPATH "RK3568 librknnrt.so link input")
if(PAIRED_BUILD_RKNN)
  find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
  if(NOT RKNN_RUNTIME)
    message(FATAL_ERROR "Explicit RKNN_RUNTIME link input required; loaded bytes are rechecked at runtime")
  endif()
  add_executable(paired_bench main.cpp ${PAIRED_PLATFORM_BINDING})
  target_link_libraries(paired_bench PRIVATE Qt${QT_VERSION_MAJOR}::Core Threads::Threads ${RKNN_RUNTIME})
endif()
